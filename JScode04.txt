function emailCheck() {
  // תוכן השדה אימייל
  let emailCon = document.getElementById("emailAddress").value;

  // המיקום של ה- @ בכתובת המייל
  let atPlace = emailCon.indexOf("@");

  // "נגזור" את כל מה שנמצא אחרי ה- @, זוהי הסיומת של כתובת המייל
  let toCheck = emailCon.substring(atPlace + 1);

  // משתנה שמסמן האם הסיומת שהזין המשתמש נמצאת במערך הסיומות השגויות שלנו. בהתחלה, לפני שמתחילה הבדיקה- היא לא נמצאת
  let isFound = false;

  // הלולאה רצה על כל מערך הסיומות השגויות. אם הסיומת שהזין המשתמש שווה לאחת המסיומות הללו-
  // נשנה את ערך המשתנה found
  // ל- 1, כלומר הסיומת נמצאת במערך הסיומות השגויות
  for (let index = 0; index < gmailTypo.length; index++) {
    if (toCheck == gmailTypo[index]) {isFound = true;
    }
  }
}
